# Pénétromètre — dépendances runtime
#
# Tkinter est livré avec Python (stdlib). Sur Debian/Ubuntu :
#   sudo apt install python3-tk
#
# Pillow est utilisée pour le rendu PNG et la génération du PDF d'expertise.
# ÉPINGLÉE exactement : le rendu PIL est verrouillé par des golden-tests
# sensibles au moteur de police de Pillow — une dérive de version peut
# décaler l'anti-aliasing et casser les goldens (et le bundle/CI doivent
# rester reproductibles). Bump = geste délibéré : changer cette ligne PUIS
# régénérer les goldens (`python -m tests.test_render --update`) et
# revalider. NB : un correctif de sécurité Pillow impose donc un bump manuel.
Pillow==12.2.0

# cryptography est utilisée pour la vérification offline des licences
# (signatures Ed25519). PyInstaller l'embarque automatiquement via ses
# hooks intégrés (cryptography ≥ 38).
cryptography>=41,<45

# openpyxl est utilisée par l'export XLSX du tableau PRBS (menu Export
# > Tableau XLSX). C'est une dépendance optionnelle : l'app démarre
# et fonctionne sans, mais l'entrée de menu signale alors qu'elle
# n'est pas disponible.
openpyxl>=3.1,<4

# fpdf2 génère la fiche d'expertise PDF en VECTORIEL (texte sélectionnable,
# net à l'impression, fichier ~3-4× plus léger) dans `fiche_pdf.py`. Le
# dessin de section et la légende restent rendus par Pillow et collés comme
# images dans la page vectorielle (approche hybride).
fpdf2>=2.8,<3
