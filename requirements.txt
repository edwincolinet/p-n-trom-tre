# Pénétromètre — dépendances runtime
#
# Tkinter est livré avec Python (stdlib). Sur Debian/Ubuntu :
#   sudo apt install python3-tk
#
# Pillow est utilisée pour le rendu PNG et la génération du PDF d'expertise.
# Plage testée localement de Pillow 10.x à 12.x. Le code n'utilise aucune
# API dépréciée (le rendu passe par `tobytes`, pas `getdata`) → compatible
# jusqu'à la 14 sans changement. On garde une borne haute prudente.
Pillow>=10.0,<13

# cryptography est utilisée pour la vérification offline des licences
# (signatures Ed25519). PyInstaller l'embarque automatiquement via ses
# hooks intégrés (cryptography ≥ 38).
cryptography>=41,<45

# openpyxl est utilisée par l'export XLSX du tableau PRBS (menu Export
# > Tableau XLSX). C'est une dépendance optionnelle : l'app démarre
# et fonctionne sans, mais l'entrée de menu signale alors qu'elle
# n'est pas disponible.
openpyxl>=3.1,<4
